def make_row_hash(values):
    """Row ka 'fingerprint': same row ka hamesha same hash."""
    joined = "\x1f".join(values)  # \x1f = alag karne wala khaas nishan
    return hashlib.md5(joined.encode("utf-8")).hexdigest()


def read_csv_file(path):
    """CSV parho. Sab kuch text rehta hai, kuch badla nahi jata."""
    with open(path, newline="", encoding="utf-8-sig") as f:
        reader = csv.reader(f)
        header = next(reader)
        rows = []
        for line_no, row in enumerate(reader, start=2):
            if row == []:  # bilkul khali line chhor do
                continue
            if len(row) != len(header):
                raise ValueError(f"{path.name} line {line_no}: columns ka number galat hai")
            rows.append(row)
    return header, rows


def read_json_file(path):
    """JSON parho. Har record poora ek text column (raw_json) mein jata hai."""
    data = json.loads(path.read_text(encoding="utf-8"))
    if isinstance(data, dict):  # shifts.json: {"shifts": [...]}
        data = next(v for v in data.values() if isinstance(v, list))
    rows = [[json.dumps(record, ensure_ascii=False)] for record in data]
    return ["raw_json"], rows

def load_one_file(conn, path, loaded_at):
    """Ek file ko raw table mein daalo aur row counts wapas do."""
    table = path.stem  # employees.csv -> employees

    if path.suffix == ".csv":
        header, rows = read_csv_file(path)
    else:
        header, rows = read_json_file(path)

    all_columns = header + ["_source_file", "_loaded_at", "_row_hash"]
    column_sql = ", ".join(f'"{c}" TEXT' for c in all_columns)
    conn.execute(f'CREATE TABLE IF NOT EXISTS raw."{table}" ({column_sql})')
    conn.commit()

    # Har row ke aage metadata jod do
    data = [tuple(r) + (path.name, loaded_at, make_row_hash(r)) for r in rows]
    placeholders = ", ".join("?" for _ in all_columns)
    column_names = ", ".join(f'"{c}"' for c in all_columns)

    try:
        # IDEMPOTENT TRICK: pehle is file ke purane rows hatao, phir naye daalo.
        # Dono kaam ek transaction mein hain: ya dono honge, ya kuch bhi nahi.
        conn.execute(f'DELETE FROM raw."{table}" WHERE _source_file = ?', (path.name,))
        conn.executemany(
            f'INSERT INTO raw."{table}" ({column_names}) VALUES ({placeholders})', data
        )
        conn.commit()
    except Exception:
        conn.rollback()
        raise

    rows_in_table = conn.execute(
        f'SELECT COUNT(*) FROM raw."{table}" WHERE _source_file = ?', (path.name,)
    ).fetchone()[0]

    return table, len(rows), rows_in_table

def date_format_of(value):
    """Ek date ka format pehchano."""
    value = value.strip()
    if value == "":
        return "BLANK"
    if re.fullmatch(r"\d{4}-\d{2}-\d{2}", value):
        return "YYYY-MM-DD"
    if re.fullmatch(r"\d{2}/\d{2}/\d{4}", value):
        return "DD/MM/YYYY"
    if re.fullmatch(r"\d{2}-[A-Za-z]{3}-\d{4}", value):
        return "DD-Mon-YYYY"
    return "OTHER"

# ---------------------------------------------------------------- chhote helpers
def clean_text(value):
    """Aage peeche ke spaces hatao aur beech ke multiple spaces ek karo."""
    return re.sub(r"\s+", " ", value).strip()


def parse_date(value):
    """Teeno formats ko YYYY-MM-DD banao. Khali ho to None."""
    value = clean_text(value)
    if value == "":
        return None
    if re.fullmatch(r"\d{4}-\d{2}-\d{2}", value):
        return datetime.strptime(value, "%Y-%m-%d").strftime("%Y-%m-%d")
    if re.fullmatch(r"\d{2}/\d{2}/\d{4}", value):  # DD/MM/YYYY
        return datetime.strptime(value, "%d/%m/%Y").strftime("%Y-%m-%d")
    if re.fullmatch(r"\d{2}-[A-Za-z]{3}-\d{4}", value):  # DD-Mon-YYYY
        return datetime.strptime(value, "%d-%b-%Y").strftime("%Y-%m-%d")
    raise ValueError(f"Ajeeb date format mila: {value!r}")


def clean_phone(value):
    """Sirf digits rakho, phir 0307-9559808 jaisa format do (agar 11 digits hon)."""
    digits = re.sub(r"\D", "", value)
    if len(digits) == 11:
        return f"{digits[:4]}-{digits[4:]}"
    return digits  # ajeeb length ho to digits hi rehne do (neeche ginti print hogi)


def build_department_lookup(conn):
    """departments.csv se naam dhoondne ki tables banao."""
    rows = pd.read_sql_query(
        "SELECT dept_name, plant, previous_name FROM raw.departments", conn
    ).to_dict("records")

    current = {}      # (plant, naam) -> sahi naam
    old = {}          # (plant, purana naam) -> naya naam
    by_name = {}      # naam (kisi bhi plant ka) -> sahi naam
    old_by_name = {}  # purana naam (kisi bhi plant ka) -> naya naam
    for r in rows:
        plant = clean_text(r["plant"]).lower()
        name = clean_text(r["dept_name"])
        prev = clean_text(r["previous_name"])
        current[(plant, name.lower())] = name
        by_name[name.lower()] = name
        if prev:
            old[(plant, prev.lower())] = name
            old_by_name[prev.lower()] = name
    return current, old, by_name, old_by_name


def clean_department(plant, raw_department, lookup):
    """Wapas do: (sahi naam, purana naam tha? 1/0, mila? True/False)"""
    current, old, by_name, old_by_name = lookup
    key = clean_text(raw_department).lower()
    plant_key = clean_text(plant).lower()

    if (plant_key, key) in current:
        return current[(plant_key, key)], 0, True
    if (plant_key, key) in old:
        return old[(plant_key, key)], 1, True
    if key in by_name:
        return by_name[key], 0, True
    if key in old_by_name:
        return old_by_name[key], 1, True
    return clean_text(raw_department).title(), 0, False  # nahi mila: shak wali entry


def clean_row(r, lookup):
    """Ek raw row ko saaf karo."""
    department, was_renamed, found = clean_department(r["plant"], r["department"], lookup)
    return {
        "emp_id": clean_text(r["emp_id"]),
        "full_name": clean_text(r["full_name"]).title(),
        "gender": clean_text(r["gender"]).upper(),
        "date_of_birth": parse_date(r["date_of_birth"]),
        "email": clean_text(r["email"]).lower(),
        "phone": clean_phone(r["phone"]),
        "plant": clean_text(r["plant"]),
        "department": department,
        "department_was_renamed": was_renamed,
        "department_found": found,
        "designation": clean_text(r["designation"]),
        "grade": clean_text(r["grade"]).upper(),
        "employment_type": clean_text(r["employment_type"]),
        "hire_date": parse_date(r["hire_date"]),
        "separation_date": parse_date(r["separation_date"]),
        "status": clean_text(r["status"]),
        "annual_leave_entitlement": int(clean_text(r["annual_leave_entitlement"])),
        "updated_at": clean_text(r["updated_at"]),
        "raw_rowid": int(r["raw_rowid"]),
    }

def run_stage_3():
    print("Starting Optimized Stage 3: Building staging.attendance_daily...")
    conn = get_connection()
    
    df_hist = pd.read_sql("SELECT emp_id, valid_from, valid_to, dept_id FROM staging.employment_history", conn)
    df_roster = pd.read_sql("SELECT emp_id, week_start, shift_code FROM staging.shift_roster", conn)
    df_shifts = pd.read_sql("SELECT shift_code, start_time, end_time, grace_minutes FROM staging.shifts", conn)
    df_holidays = pd.read_sql("SELECT holiday_date, plant FROM staging.holidays", conn)
    df_depts = pd.read_sql("SELECT dept_id, plant FROM staging.departments", conn)
    df_leave = pd.read_sql("SELECT emp_id, leave_date FROM staging.leave_days", conn)
    df_punches = pd.read_sql("SELECT emp_id, punch_time FROM staging.punches_clean", conn)
    conn.close()
    
    print("Vectorizing calendar and schedule generation...")
    df_hist['valid_from'] = pd.to_datetime(df_hist['valid_from'])
    df_hist['valid_to'] = pd.to_datetime(df_hist['valid_to'])
    
    all_dates = pd.date_range(start='2026-04-01', end='2026-09-30')
    
    rows = []
    for _, h in df_hist.iterrows():
        d_range = all_dates[(all_dates >= h['valid_from']) & (all_dates <= h['valid_to'])]
        if not d_range.empty:
            temp = pd.DataFrame({'emp_id': h['emp_id'], 'work_date': d_range, 'dept_id': h['dept_id']})
            rows.append(temp)
            
    df_sched = pd.concat(rows, ignore_index=True)
    df_sched['work_date_str'] = df_sched['work_date'].dt.strftime('%Y-%m-%d')
    df_sched['weekday'] = df_sched['work_date'].dt.day_of_week
    
    dept_plant = df_depts.set_index('dept_id')['plant'].to_dict()
    df_sched['plant'] = df_sched['dept_id'].map(dept_plant).fillna('Head Office')
    
    df_sched['monday'] = df_sched['work_date'] - pd.to_timedelta(df_sched['weekday'], unit='D')
    df_roster['week_start'] = pd.to_datetime(df_roster['week_start'])
    
    df_sched = df_sched.sort_values('work_date')
    df_roster = df_roster.sort_values('week_start')
    
    df_merged = pd.merge_asof(
        df_sched, 
        df_roster, 
        left_on='monday', 
        right_on='week_start', 
        by='emp_id', 
        direction='backward'
    )
    df_merged['shift_code'] = df_merged['shift_code'].fillna('GEN')
    
    df_punches['punch_dt'] = pd.to_datetime(df_punches['punch_time'])
    df_punches['work_date_str'] = df_punches['punch_dt'].dt.strftime('%Y-%m-%d')
    punch_agg = df_punches.groupby(['emp_id', 'work_date_str']).agg(
        check_in=('punch_dt', 'min'),
        check_out=('punch_dt', 'max'),
        punch_count=('punch_dt', 'count')
    ).reset_index()
    
    df_merged = pd.merge(df_merged, punch_agg, on=['emp_id', 'work_date_str'], how='left')
    
    holiday_set = set(zip(df_holidays['holiday_date'], df_holidays['plant']))
    holiday_all = set(df_holidays[df_holidays['plant'] == 'All']['holiday_date'])
    leave_set = set(zip(df_leave['emp_id'], df_leave['leave_date']))
    shift_dict = df_shifts.set_index('shift_code').to_dict('index')
    
    print("Calculating attendance status, late minutes and overtime...")
    statuses = []
    worked_hours = []
    late_mins = []
    early_mins = []
    overtime_hrs = []
    cin_list = []
    cout_list = []
    
    for _, row in df_merged.iterrows():
        d_str = row['work_date_str']
        wday = row['weekday']
        emp = row['emp_id']
        plt = row['plant']
        sc = row['shift_code']
        
        is_sunday = (wday == 6)
        is_hol = ((d_str, plt) in holiday_set) or (d_str in holiday_all)
        is_leave = ((emp, d_str) in leave_set)
        
        pc = row['punch_count']
        cin = row['check_in']
        cout = row['check_out']
        
        status = 'Absent'
        wh = 0.0
        lm = 0.0
        em = 0.0
        ot = 0.0
        cin_str = None
        cout_str = None
        
        if is_sunday:
            status = 'Weekly Off'
        elif is_hol:
            status = 'Holiday'
        elif is_leave:
            status = 'On Leave'
        elif pd.notna(pc) and pc >= 2:
            status = 'Present'
            cin_str = cin.strftime('%Y-%m-%d %H:%M:%S')
            cout_str = cout.strftime('%Y-%m-%d %H:%M:%S')
            
            wh = (cout - cin).total_seconds() / 3600.0
            s_info = shift_dict.get(sc, {'start_time': '09:00', 'end_time': '17:00', 'grace_minutes': 10})
            
            st_parts = str(s_info.get('start_time', '09:00')).split(':')
            et_parts = str(s_info.get('end_time', '17:00')).split(':')
            
            s_dt = cin.replace(hour=int(st_parts[0]), minute=int(st_parts[1]), second=0, microsecond=0)
            e_dt = cout.replace(hour=int(et_parts[0]), minute=int(et_parts[1]), second=0, microsecond=0)
            
            grace = float(s_info.get('grace_minutes', 10) or 10)
            diff_m = (cin - s_dt).total_seconds() / 60.0
            if diff_m > grace:
                lm = diff_m - grace
                
            ediff = (e_dt - cout).total_seconds() / 60.0
            if ediff > 10:
                em = ediff
                
            if cout > e_dt:
                ot_m = (cout - e_dt).total_seconds() / 60.0
                ot = (ot_m // 30) * 0.5
        elif pd.notna(pc) and pc == 1:
            status = 'Missing Punch'
            cin_str = cin.strftime('%Y-%m-%d %H:%M:%S')
            cout_str = cin_str
        else:
            status = 'Absent'
            
        statuses.append(status)
        worked_hours.append(round(wh, 2))
        late_mins.append(round(lm, 2))
        early_mins.append(round(em, 2))
        overtime_hrs.append(round(ot, 2))
        cin_list.append(cin_str)
        cout_list.append(cout_str)
        
    df_merged['status'] = statuses
    df_merged['worked_hours'] = worked_hours
    df_merged['late_minutes'] = late_mins
    df_merged['early_minutes'] = early_mins
    df_merged['overtime_hours'] = overtime_hrs
    df_merged['check_in'] = cin_list
    df_merged['check_out'] = cout_list
    
    final_df = df_merged[['emp_id', 'work_date_str', 'shift_code', 'status', 'check_in', 'check_out', 'worked_hours', 'late_minutes', 'early_minutes', 'overtime_hours']]
    final_df.columns = ['emp_id', 'work_date', 'shift_code', 'status', 'check_in', 'check_out', 'worked_hours', 'late_minutes', 'early_minutes', 'overtime_hours']
    
    print(f"Total attendance daily rows generated: {len(final_df)}")
    
    # Save using Temp Table Strategy for attached SQLite databases
    conn = get_connection()
    conn.execute("DROP TABLE IF EXISTS staging.attendance_daily;")
    conn.execute("""
        CREATE TABLE staging.attendance_daily (
            emp_id TEXT,

            shift_code TEXT,
            status TEXT,
            check_in TEXT,
            check_out TEXT,
            worked_hours REAL,
            late_minutes REAL,
            early_minutes REAL,
            overtime_hours REAL
        );
    """)
    conn.commit()
    
    # Write dataframe to temp table first
    final_df.to_sql("attendance_daily_temp", conn, index=False, if_exists="replace", chunksize=10000)
    
    # Insert from temp table into staging schema table
    conn.execute("""
        INSERT INTO staging.attendance_daily 
        SELECT emp_id, work_date, shift_code, status, check_in, check_out, worked_hours, late_minutes, early_minutes, overtime_hours 
        FROM attendance_daily_temp;
    """)
    conn.execute("DROP TABLE attendance_daily_temp;")
    conn.commit()
    
    att_count = conn.execute("SELECT COUNT(*) FROM staging.attendance_daily").fetchone()[0]
    
    print(f"\n================ STAGE 3 RESULT ================")
    print(f"Staging Attendance Daily rows: {att_count}")
    
    if att_count > 0:
        print("STATUS: PASS ✅")
    else:
        print("STATUS: RUK JAO 🛑 - Table is empty!")
        sys.exit(1)
        
    conn.close()
